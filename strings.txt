let currentPhase = 'jamo';
let currentGroupIdx = 0;
let currentItem = null;

let jMaxIdx = -1;
let sMaxIdx = -1;

function normalise(s) { return s.trim().toLowerCase().replace(/\s+/g, ''); }

function getQuestion(phase, groupName) {
    const data = phase === 'jamo' ? jData : sData;
    const candidates = data.filter(d => d[3] === groupName); // index 3 is group_name in data tuples
    if (candidates.length === 0) return null;
    return candidates[Math.floor(Math.random() * candidates.length)];
}

function renderQuestion() {
    const groups = currentPhase === 'jamo' ? jGroups : sGroups;
    if (currentGroupIdx >= groups.length) {
        nextPhase();
        return;
    }
    
    document.getElementById('phase-label').innerText = currentPhase.toUpperCase() + ' - Livello ' + (currentGroupIdx + 1);
    
    currentItem = getQuestion(currentPhase, groups[currentGroupIdx]);
    if (!currentItem) {
        nextPhase();
        return;
    }
    
    document.getElementById('char-display').innerText = currentItem[0]; // front
    const inp = document.getElementById('answer-input');
    inp.value = '';
    inp.focus();
}

function nextPhase() {
    if (currentPhase === 'jamo') {
        currentPhase = 'syllable';
        currentGroupIdx = 0;
        renderQuestion();
    } else {
        finishTest();
    }
}

function submitAnswer(attempted) {
    if (!currentItem) return;
    
    const inp = document.getElementById('answer-input').value;
    const isCorrect = attempted && (normalise(inp) === normalise(currentItem[1]) || normalise(inp) === normalise(currentItem[2]));
    
    if (isCorrect) {
        if (currentPhase === 'jamo') jMaxIdx = currentGroupIdx;
        if (currentPhase === 'syllable') sMaxIdx = currentGroupIdx;
        currentGroupIdx++;
        renderQuestion();
    } else {
        // Failed or "I don't know" -> stop phase
        nextPhase();
    }
}

/**
   * Compute a percentage fill for the track gradient.
   * @param {HTMLInputElement} input
   */
  function computeFillPct(input) {
    const min  = parseFloat(input.min)  || 0;
    const max  = parseFloat(input.max)  || 100;
    const val  = parseFloat(input.value);
    return ((val - min) / (max - min)) * 100;
  }

  /**
   * Update the display label and track fill on range change.
   * @param {HTMLInputElement} input   - The range element
   * @param {string}           labelId - ID of the value display element
   * @param {string}           unit    - Unit suffix ("min" | "carte")
   * @param {number[]}         presets - List of preset values to sync active state
   */
  function updateRange(input, labelId, unit, presets) {
    const val = parseFloat(input.value);
    const pct = computeFillPct(input);

    // Update track fill
    input.style.setProperty('--fill-pct', pct + '%');

    // Update ARIA
    input.setAttribute('aria-valuenow', val);

    // Update display badge
    const label = document.getElementById(labelId);
    if (label) {
      label.textContent = val + ' ' + unit;
    }

    // Sync preset button active states
    syncPresets(input.id, val, presets);
  }

  /**
   * Sync which preset button appears active.
   * @param {string}   inputId - ID of the range input
   * @param {number}   val     - Current value
   * @param {number[]} presets - List of preset values
   */
  function syncPresets(inputId, val, presets) {
    const input = document.getElementById(inputId);
    if (!input) return;
    const fieldRow = input.closest('.field-row');
    if (!fieldRow) return;
    const btns = fieldRow.querySelectorAll('.preset-btn');
    btns.forEach(btn => {
      const btnVal = parseFloat(btn.getAttribute('data-val'));
      btn.classList.toggle('active', btnVal === val);
    });
  }

  /**
   * Click on a preset button: update range + display.
   * @param {string}   inputId - ID of the range input
   * @param {number}   value   - Value to set
   * @param {string}   labelId - ID of the display label
   * @param {string}   unit    - Unit suffix
   * @param {number[]} presets - List of preset values
   */
  function setPreset(inputId, value, labelId, unit, presets) {
    const input = document.getElementById(inputId);
    if (!input) return;
    input.value = value;
    updateRange(input, labelId, unit, presets);
  }

  /**
   * Initialise range fills and preset active states on page load.
   */
  function initRanges() {
    const configs = [
      { inputId: 'range-minutes', labelId: 'display-minutes', unit: 'min',   presets: [10, 15, 20, 30] },
      { inputId: 'range-cards',   labelId: 'display-cards',   unit: 'carte', presets: [3, 5, 8, 10]    },
    ];

    configs.forEach(({ inputId, labelId, unit, presets }) => {
      const input = document.getElementById(inputId);
      if (!input) return;
      input.style.setProperty('--fill-pct', computeFillPct(input) + '%');
      syncPresets(inputId, parseFloat(input.value), presets);
    });
  }

  /**
   * Auto-dismiss the success toast after 4 s.
   */
  function initToast() {
    const toast = document.querySelector('.toast-success');
    if (!toast) return;
    setTimeout(() => {
      toast.style.transition = 'opacity 0.5s ease, transform 0.5s ease';
      toast.style.opacity = '0';
      toast.style.transform = 'translateY(-8px)';
      setTimeout(() => toast.remove(), 500);
    }, 4000);
  }

  document.addEventListener('DOMContentLoaded', () => {
    initRanges();
    initToast();
  });

let _cardStart   = Date.now();
let _answered    = false;
let _wasCorrect  = false;
let _pauseStart  = null;

function showPause() {
  document.getElementById('pause-overlay').classList.add('visible');
  _pauseStart = Date.now();
}
function hidePause() {
  document.getElementById('pause-overlay').classList.remove('visible');
  if (_pauseStart !== null) {
    _cardStart += (Date.now() - _pauseStart);
    _pauseStart = null;
  }
}

document.addEventListener('visibilitychange', () => {
  document.hidden ? showPause() : hidePause();
});
window.addEventListener('blur', showPause);
window.addEventListener('focus', hidePause);

function normalise(s) {
  return s.trim().toLowerCase().replace(/\s+/g, '');
}

function isMatch(input) {
  const n = normalise(input);
  if (!n) return false;
  const targets = [];
  
  if (CORRECT_ANSWER) {
    targets.push(normalise(CORRECT_ANSWER));
    if (CORRECT_ANSWER.includes('/')) {
      CORRECT_ANSWER.split('/').forEach(t => targets.push(normalise(t)));
    }
  }
  
  if (CORRECT_ROMANJI) {
    targets.push(normalise(CORRECT_ROMANJI));
    if (CORRECT_ROMANJI.includes('/')) {
      CORRECT_ROMANJI.split('/').forEach(t => targets.push(normalise(t)));
    }
  }
  
  return targets.includes(n);
}

function checkAnswer() {
  if (_answered) { submitQuality(_wasCorrect ? 5 : 0); return; }

  const input = document.getElementById('answer-input').value;
  if (!input.trim()) { document.getElementById('answer-input').focus(); return; }

  _answered   = true;
  _wasCorrect = isMatch(input);

  const inp = document.getElementById('answer-input');
  inp.disabled = true;
  inp.classList.add(_wasCorrect ? 'correct' : 'wrong');

  document.getElementById('answer-reveal').style.display = 'block';

  const panel = document.getElementById('feedback-panel');
  const ftxt  = document.getElementById('feedback-text');
  const fhint = document.getElementById('feedback-hint');
  panel.style.display = 'block';

  if (_wasCorrect) {
    panel.className = 'ok';
    ftxt.textContent = '✅ Corretto!';
    fhint.textContent = CORRECT_ANSWER + (CORRECT_ROMANJI ? ' — ' + CORRECT_ROMANJI : '');
    setTimeout(() => { if (_wasCorrect) submitQuality(5); }, 1200);
  } else {
    panel.className = 'bad';
    ftxt.textContent = '❌ Risposta: ' + CORRECT_ANSWER;
    fhint.textContent = (CORRECT_ROMANJI ? CORRECT_ROMANJI + ' — ' : '') + 'Hai scritto: "' + input + '"';
  }

  document.getElementById('quality-row').style.display = 'block';

  const btn = document.getElementById('check-btn');
  btn.textContent = '→ Avanti';
}

function submitQuality(q) {
  if (!_wasCorrect && q > 2) q = 1;
  const elapsed = Date.now() - _cardStart - (_pauseStart ? Date.now() - _pauseStart : 0);
  document.getElementById('quality-input').value = q;
  document.getElementById('time-ms-input').value = elapsed;
  document.getElementById('review-form').submit();
}

document.addEventListener('keydown', function(e) {
  if (!_answered) {
    if (e.key === 'Enter') { e.preventDefault(); checkAnswer(); }
  } else {
    if (e.key === 'Enter') { e.preventDefault(); submitQuality(_wasCorrect ? 5 : 1); }
    if (e.key === '1') submitQuality(0);
    if (e.key === '2') submitQuality(1);
    if (e.key === '3') submitQuality(3);
    if (e.key === '4') submitQuality(5);
  }
});

const CORRECT_ANSWER  = {{ card.back | tojson }};
const CORRECT_ROMANJI = {{ (card.romanji or '') | tojson }};

let _answered   = false;
let _wasCorrect = false;
let _autoTimer  = null;
let _pauseStart = null;
let _studyDone  = false;

// ── Audio Helper ──────────────────────────────────────────────
function speakText(text) {
  if (!window.speechSynthesis) return;
  window.speechSynthesis.cancel();
  const utt = new SpeechSynthesisUtterance(text);
  utt.lang = 'ko-KR';
  utt.rate = 0.85;
  setTimeout(() => window.speechSynthesis.speak(utt), 100);
}

// ── Pause overlay ─────────────────────────────────────────────────
function showPause() {
  document.getElementById('pause-overlay').classList.add('visible');
  _pauseStart = Date.now();
  if (_autoTimer) { clearTimeout(_autoTimer); _autoTimer = null; }
}
function hidePause() {
  document.getElementById('pause-overlay').classList.remove('visible');
  if (_pauseStart !== null) {
    const gap = Date.now() - _pauseStart;
    _cardStart += gap;
    if (_quizStart !== null) _quizStart += gap;
    _pauseStart = null;
  }
  if (_studyDone && !_answered) {
    const inp = document.getElementById('quiz-input');
    if (inp) inp.focus();
  }
}

document.addEventListener('visibilitychange', () => {
  document.hidden ? showPause() : hidePause();
});
window.addEventListener('blur',  showPause);
window.addEventListener('focus', hidePause);

// ── Normalise for comparison ──────────────────────────────────────
function normalise(s) {
  return s.trim().toLowerCase()
    .replace(/\s+/g, '');
}

function isMatch(input) {
  const n = normalise(input);
  if (!n) return false;
  const targets = [];
  
  if (CORRECT_ANSWER) {
    targets.push(normalise(CORRECT_ANSWER));
    if (CORRECT_ANSWER.includes('/')) {
      CORRECT_ANSWER.split('/').forEach(t => targets.push(normalise(t)));
    }
  }
  
  if (CORRECT_ROMANJI) {
    targets.push(normalise(CORRECT_ROMANJI));
    if (CORRECT_ROMANJI.includes('/')) {
      CORRECT_ROMANJI.split('/').forEach(t => targets.push(normalise(t)));
    }
  }
  
  return targets.includes(n);
}

// ── Switch from study to quiz ─────────────────────────────────────
function startQuiz() {
  if (_studyDone) return;
  _studyDone = true;
  if (_autoTimer) { clearTimeout(_autoTimer); _autoTimer = null; }

  document.getElementById('study-phase').style.display = 'none';
  document.getElementById('quiz-phase').style.display  = 'block';
  
  document.getElementById('quiz-input').focus();
  speakText('{{ card.front }}'); // Auto-play when quiz starts

  _quizStart = Date.now();

  const inp = document.getElementById('quiz-input');
  inp.focus();
}

// ── Check answer ──────────────────────────────────────────────────
function checkAnswer() {
  if (_answered) return;

  const inp = document.getElementById('quiz-input');
  const val = inp.value;
  if (!val.trim()) { inp.focus(); return; }

  _answered   = true;
  _wasCorrect = isMatch(val);

  inp.disabled = true;
  inp.classList.add(_wasCorrect ? 'correct' : 'wrong');
  document.getElementById('verify-btn').disabled = true;

  const panel = document.getElementById('feedback-panel');
  const fMain = document.getElementById('feedback-main');
  const fSub  = document.getElementById('feedback-sub');
  panel.style.display = 'block';

  if (_wasCorrect) {
    panel.className    = 'ok';
    fMain.textContent  = '\u2705 Corretto!';
    fSub.textContent   = CORRECT_ANSWER + (CORRECT_ROMANJI ? ' \u2014 ' + CORRECT_ROMANJI : '');
    // Auto-advance after 0.8s on correct
    setTimeout(submitLearn, 800);
  } else {
    document.getElementById('fast_lane_failed').value = "true";
    panel.className   = 'bad';
    fMain.textContent = '\u274C Risposta: ' + CORRECT_ANSWER;
    fSub.textContent  = (CORRECT_ROMANJI ? CORRECT_ROMANJI + ' \u2014 ' : '') + 'Hai scritto: "' + val + '"';
    // Show continue button
    document.getElementById('continue-btn').style.display = 'block';
  }
}

// ── Submit learn result ────────────────────────────────────────────
function submitLearn() {
  const now     = Date.now();
  const elapsed = now - (_quizStart || _cardStart) -
                  (_pauseStart ? now - _pauseStart : 0);
  document.getElementById('time-ms-input').value = elapsed;
  document.getElementById('learn-form').submit();
}

// ── Keyboard shortcuts ─────────────────────────────────────────────
document.addEventListener('keydown', function(e) {
  if (!_studyDone) {
    if (e.key === ' ' || e.key === 'Enter') {
      e.preventDefault();
      startQuiz();
    }
    return;
  }

  if (!_answered) {
    if (e.key === 'Enter') {
      e.preventDefault();
      checkAnswer();
    }
    return;
  }

  if (e.key === 'Enter') {
    e.preventDefault();
    submitLearn();
  }
});
